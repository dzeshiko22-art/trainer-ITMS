Связка с Google Таблицей (Ведомость экзаменов)

После завершения любой попытки экзамена (как успешной, так и несданной с 1-го раза) тренажер автоматически отправляет в вашу Google Таблицу: Дату и время, ФИО сотрудника, Город, Номер попытки, Статус (Сдан / Не сдан), Баллы (из 8), Очки в игре и Построенные ступеньки.

💡 Чтобы ссылка автоматически работала у всех сотрудников без ручного ввода, вставьте полученный URL в константу DEFAULT_GOOGLE_SHEETS_URL в начале скрипта файла index.html.
📖 Как подключить свою Google Таблицу за 2 минуты (универсальный скрипт GET + POST)
1. Создайте новую Google Таблицу.
2. В меню откройте Расширения → Apps Script.
3. Удалите весь код в редакторе, вставьте обновленный код ниже и нажмите Сохранить:
function writeRowToSheet(data) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow([ "Дата и время", "ФИО сотрудника", "Город / ТТ", "Попытка №", "Статус экзамена", "Правильных ответов", "Всего вопросов", "Процент", "Монеты в игре", "Ступеньки успеха" ]); } sheet.appendRow([ data.timestamp || new Date().toLocaleString("ru-RU"), data.fullName || "", data.city || "", data.attempt || 1, data.status || "", data.score || 0, data.total || 8, data.percent || "0%", data.coins || 0, data.stairs || "0/10" ]); } function doGet(e) { if (e && e.parameter && e.parameter.fullName) { writeRowToSheet(e.parameter); } return ContentService .createTextOutput(JSON.stringify({ result: "success" })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var data = {}; if (e && e.postData && e.postData.contents) { data = JSON.parse(e.postData.contents); } else if (e && e.parameter) { data = e.parameter; } writeRowToSheet(data); return ContentService .createTextOutput(JSON.stringify({ result: "success" })) .setMimeType(ContentService.MimeType.JSON); }

4. Нажмите синюю кнопку Начать развертывание → Новое развертывание (если развертывание уже было — выберите Управление развертываниями → ✏️ → Новая версия!).
5. Тип: Веб-приложение. Запуск от имени: От моего имени. Доступ: Все (Anyone).
6. Нажмите Развернуть и вставьте ссылку .../exec в поле выше.
